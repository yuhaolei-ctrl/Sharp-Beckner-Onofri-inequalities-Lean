def prefixes : List String := ["", "BecknerOnofri.PaperResult.", "BecknerOnofri.Target.", "BecknerOnofri.Paper2.",
  "BecknerOnofri.HighDim.", "BecknerOnofri."]

def tsvEsc (s : String) : String := (s.replace "\t" " ").replace "\n" " "

run_cmd do
  let env <- getEnv
  let mut out := "candidate\tstatus\tfull_name\tkind\ttype\n"
  for c in candidates do
    let mut found : Array Name := #[]
    for p in prefixes do
      let n := (p ++ c).toName
      if env.contains n && !found.contains n then found := found.push n
    if found.isEmpty then
      out := out ++ s!"{c}\tNOT_FOUND\t\t\t\n"
    else
      for n in found do
        let ci := env.find? n
        let kind := match ci with
          | some (.thmInfo _) => "theorem" | some (.defnInfo _) => "def" | some (.axiomInfo _) => "AXIOM"
          | some (.inductInfo _) => "inductive" | some (.opaqueInfo _) => "opaque" | _ => "other"
        let ty <- liftCoreM (MetaM.run' (do
          match ci with
          | some i => return toString (← ppExpr i.type)
          | none => return ""))
        out := out ++ s!"{c}\tFOUND\t{n}\t{kind}\t{tsvEsc ty}\n"
  IO.FS.writeFile "/home/user/v2work/logs/G00_candidate_resolution.tsv" out
